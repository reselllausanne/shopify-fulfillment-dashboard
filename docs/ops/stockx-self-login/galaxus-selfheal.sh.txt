#!/bin/bash
# Host-side ops loop (not repo code). Keeps the Galaxus StockX account logged in WITHOUT human help.
# Every 20 min: token <3h left or login dead -> remint through a fresh Lausanne session; if the Auth0 session is dead ->
# parallel throwaway-profile login (Turnstile click + email OTP), then activate (session file + profile cookies + token).
C=resell-web-1
D=/opt/resell/.data
LOG=$D/galaxus-selfheal.sup.log
log() { echo "$(date -u +%FT%TZ) $*" >> $LOG; }
ready() { docker exec $C true 2>/dev/null && [ "$(docker inspect -f '{{.State.StartedAt}}' $C | xargs -I{} date -u -d {} +%s)" -lt "$(( $(date -u +%s) - 30 ))" ]; }
left_h() { docker exec $C node -e '
const j=JSON.parse(require("fs").readFileSync("/app/.data/stockx-token-galaxus.json","utf8"));
const t=j.token||"";try{const e=JSON.parse(Buffer.from(t.split(".")[1].replace(/-/g,"+").replace(/_/g,"/"),"base64").toString()).exp*1000;console.log(Math.floor((e-Date.now())/3600000));}catch{console.log(-1)}' 2>/dev/null; }
restart_buyer() {
  docker exec -w /app $C sh -c 'rm -f /app/.data/night-buyer.halt; npx tsx scripts/stockx-buyer.ts resume >/dev/null 2>&1; npx tsx scripts/stockx-buyer.ts start >/dev/null 2>&1'
  docker exec -d -w /app $C sh -c 'set -a; . /app/.data/stockx-buyer/buyer.env; set +a; npx tsx scripts/stockx-buyer.ts run --force >> /app/.data/stockx-buyer.log 2>&1'
}
stop_buyer() {
  docker exec -w /app $C npx tsx scripts/stockx-buyer.ts stop >/dev/null 2>&1
  for j in $(seq 1 60); do docker exec $C sh -c 'ps -eo args | grep -q "[s]tockx-buyer.ts run"' || break; sleep 5; done
}
log "selfheal start"
while true; do
  sleep 1200
  ready || continue
  h=$(left_h); h=${h:-0}
  [ "$h" -ge 3 ] && continue
  log "token left ${h}h -> remint"
  docker cp /root/galaxus-login/_tmp162.ts $C:/app/scripts/_tmp162.ts
  docker exec -w /app $C sh -c 'npx tsx scripts/_tmp162.ts >> /app/.data/galaxus-selfheal.err 2>&1; rm -f scripts/_tmp162.ts'
  res=$(grep -E "RESULT" $D/galaxus-selfheal.log | tail -1)
  log "remint: $res"
  echo "$res" | grep -q "REMINT_OK" && continue
  echo "$res" | grep -q "auth0_session_dead" || continue
  log "Auth0 session dead -> autonomous login"
  stop_buyer
  rm -f $D/galaxus-login.proven $D/galaxus-profile.proven; rm -rf $D/galaxus-login.lock
  for n in 1 2 3 4 5 6; do
    ready || { sleep 30; continue; }
    docker cp /root/galaxus-login/_tmp159.ts $C:/app/scripts/_tmp159.ts
    docker exec -w /app $C sh -c 'npx tsx scripts/_tmp159.ts >> /app/.data/galaxus-login.err 2>&1; rm -f scripts/_tmp159.ts'
    [ -f $D/galaxus-login.proven ] && break
  done
  if [ -f $D/galaxus-login.proven ]; then
    docker cp /root/galaxus-login/_tmp161.ts $C:/app/scripts/_tmp161.ts
    docker exec -w /app $C sh -c 'npx tsx scripts/_tmp161.ts >> /app/.data/galaxus-activate.err 2>&1; rm -f scripts/_tmp161.ts'
    log "login+activate done: $(tail -1 $D/galaxus-activate.log)"
  else
    log "autonomous login FAILED after 6 runs (see galaxus-login.log)"
  fi
  restart_buyer
done
